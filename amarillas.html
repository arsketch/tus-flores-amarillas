    <div id="text-container"></div>

    <div id="counter"></div>


    <script>

        // Contador

        function updateCounter() {

            const start = new Date("2021-12-08T00:11:00");

            const now = new Date();

            const diff = now - start;

            const days = Math.floor(diff / (1000 * 60 * 60 * 24));

            const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));

            const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));

            document.getElementById('counter').innerText = `${days} días, ${hours} horas y ${minutes} minutos juntos.`;

        }

        setInterval(updateCounter, 60000);

        updateCounter();


        // Interacción

        function showBouquet() {

            const container = document.getElementById('canvas-container');

            const flower = document.getElementById('flower');

            if (flower.innerText === '🌻') {

                flower.innerText = '💐';

                container.classList.add('slide-right');

                createHearts();

                typeWriter("Cada día lejos vale la pena por ti. Te amo infinitamente.", 0);

            }

        }


        // Máquina de escribir

        function typeWriter(text, i) {

            const container = document.getElementById('text-container');

            container.style.opacity = 1;

            if (i < text.length) {

                container.innerHTML += text.charAt(i);

                setTimeout(() => typeWriter(text, i + 1), 50);

            }

        }


        // Lluvia de corazones

        function createHearts() {

            setInterval(() => {

                const heart = document.createElement('div');

                heart.innerHTML = '💛';

                heart.className = 'heart';

                heart.style.left = Math.random() * 100 + 'vw';

                heart.style.animationDuration = Math.random() * 3 + 2 + 's';

                document.body.appendChild(heart);

                setTimeout(() => heart.remove(), 5000);

            }, 300);

        }

    </script>
